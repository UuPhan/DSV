const input = $json.output || $json;


// ============================================================
// 1. Validate input
// ============================================================

const run_id = String(
  input.run_id || ''
).trim();

if (!run_id) {
  throw new Error(
    'run_id is missing.'
  );
}


const variant = String(
  input.variant || ''
).trim();

if (!variant) {
  throw new Error(
    'variant is missing.'
  );
}


const mode =
  String(
    input.mode || ''
  ).trim().toLowerCase() || null;


// ============================================================
// 2. Resolve max duration
// ============================================================
//
// Normal:
//   30 seconds
//
// Full:
//   170 seconds
//
// Prefer explicit input.max_duration.
// Otherwise derive from mode.
//

const max_duration = Number(
  input.max_duration ??
  (
    mode === 'full'
      ? 170
      : 30
  )
);

if (
  !Number.isFinite(max_duration) ||
  max_duration <= 0
) {
  throw new Error(
    `Invalid max_duration: ${max_duration}`
  );
}


// ============================================================
// 3. Validate caption
// ============================================================

const caption = String(
  input.caption || ''
).trim();

if (!caption) {
  throw new Error(
    'caption is missing.'
  );
}


// ============================================================
// 4. Validate lyrics
// ============================================================

const lyrics = String(
  input.lyrics || ''
).trim();

if (!lyrics) {
  throw new Error(
    'lyrics is missing.'
  );
}

// 5. Generate seed
// ============================================================
