const fs = require('fs');

const items = $input.all();

const DEFAULT_PRODUCTION_ROOT = '/data/productions';


/* =========================================================
 * Helpers
 * =======================================================*/

function cleanString(value) {
  if (value === null || value === undefined) {
    return '';
  }

  return String(value).trim();
}


function nullableString(value) {
  const valueClean = cleanString(value);
  return valueClean !== '' ? valueClean : null;
}


function safePart(value) {
  return cleanString(value)
    .replace(/[^a-zA-Z0-9._-]+/g, '_')
    .replace(/^_+|_+$/g, '');
}


/*
 * IMPORTANT:
 * Preserve the leading "/" when the first path is absolute.
 *
 * /data/productions
 * must NOT become
 * data/productions
 */
function joinPath(...parts) {
  if (!parts.length) {
    return '';
  }

  const first = String(parts[0] ?? '');

  const isAbsolute = first.startsWith('/');

  const cleaned = parts
    .map(part =>
      String(part ?? '')
        .replace(/^\/+|\/+$/g, '')
    )
    .filter(Boolean)
    .join('/');

  return isAbsolute
    ? `/${cleaned}`
    : cleaned;
}


function ensureDir(dirPath) {
  fs.mkdirSync(dirPath, {
    recursive: true
  });
}


function readJsonFile(filePath, fallback) {
  if (!fs.existsSync(filePath)) {
    return fallback;
  }

  try {
    const raw = fs.readFileSync(filePath, 'utf8');

    if (!raw.trim()) {
      return fallback;
    }

    return JSON.parse(raw);
  } catch (error) {
    throw new Error(
      `Save Artifact: failed to read JSON file "${filePath}": ${error.message}`
    );
  }
}


function writeJsonFile(filePath, data) {
  fs.writeFileSync(
    filePath,
    JSON.stringify(data, null, 2),
    'utf8'
  );
}


function normalizeReference(reference) {
  if (reference === null || reference === undefined) {
    return [];
  }

  if (Array.isArray(reference)) {
    return reference
      .map(value => cleanString(value))
      .filter(Boolean);
  }

  const value = cleanString(reference);

  return value ? [value] : [];
}


function getBinaryExtension(binaryInfo) {
  const extension = cleanString(
    binaryInfo.fileExtension ||
    binaryInfo.extension ||
    ''
  ).toLowerCase();

  if (extension) {
    return extension.replace(/^\./, '');
  }

  const fileName = cleanString(
    binaryInfo.fileName ||
    binaryInfo.filename ||
    ''
  );

  const match = fileName.match(/\.([a-zA-Z0-9]+)$/);

  if (match) {
    return match[1].toLowerCase();
  }

  return '';
}


function getMimeType(binaryInfo) {
  return cleanString(
    binaryInfo.mimeType ||
    binaryInfo.mime_type ||
    'application/octet-stream'
  );
}


function resolveBinaryInfo(item, binaryField) {
  const binary = item.binary || {};

  if (!binary[binaryField]) {
    throw new Error(
      `Save Artifact: binary field "${binaryField}" not found`
    );
  }

  return binary[binaryField];
}


/* =========================================================
 * Main
 * =======================================================*/

const output = [];


for (
  let itemIndex = 0;
  itemIndex < items.length;
  itemIndex++
) {
  const item = items[itemIndex];

  const input = item.json || {};


  /* -------------------------------------------------------
   * Basic fields
   * -----------------------------------------------------*/

  const run_id = cleanString(
    input.run_id ??
    input.output?.run_id
  );

  if (!run_id) {
    throw new Error(
      'Save Artifact: missing run_id'
    );
  }


  const productionRoot =
    cleanString(input.production_root) ||
    DEFAULT_PRODUCTION_ROOT;


  const runDir =
    cleanString(input.run_dir) ||
    joinPath(
      productionRoot,
      run_id
    );


  const artifact_name = safePart(
    input.artifact_name ||
    input.name
  );

  if (!artifact_name) {
    throw new Error(
      'Save Artifact: missing artifact_name'
    );
  }


  /*
   * variant:
   *
   * null for lyrics/image/video/etc.
   * "00" for music variants.
   */
  const variant = nullableString(
    input.variant
  );


  /*
   * mode:
   *
   * null
   * preview
   * full
   */
  const modeRaw = nullableString(
    input.mode
  );

  const mode = modeRaw
    ? modeRaw.toLowerCase()
    : null;


  /*
   * Binary field.
   *
   * Default is "data".
   */
  const binaryField =
    cleanString(
      input.binary_field
    ) || 'data';


  /* -------------------------------------------------------
   * Manifest
   * -----------------------------------------------------*/

  const manifestPath =
    cleanString(input.manifest_path) ||
    joinPath(
      runDir,
      `${run_id}.artifacts.json`
    );


  ensureDir(runDir);


  let manifestData = readJsonFile(
    manifestPath,
    {
      run_id,
      artifacts: []
    }
  );


  if (
    !manifestData ||
    typeof manifestData !== 'object'
  ) {
    manifestData = {
      run_id,
      artifacts: []
    };
  }


  if (!Array.isArray(manifestData.artifacts)) {
    manifestData.artifacts = [];
  }


  if (!manifestData.run_id) {
    manifestData.run_id = run_id;
  }


  /* -------------------------------------------------------
   * Determine next version
   *
   * Scope:
   *   name + variant + mode
   *
   * Therefore:
   *
   * audio / 00 / preview
   * audio / 00 / full
   *
   * have independent version sequences.
   * -----------------------------------------------------*/

  const scopedArtifacts =
    manifestData.artifacts.filter(
      artifact =>
        cleanString(artifact?.name) === artifact_name &&
        nullableString(artifact?.variant) === variant &&
        nullableString(artifact?.mode) === mode
    );


  let maxVersion = 0;


  for (const artifact of scopedArtifacts) {
    const versionNumber =
      Number(artifact?.version);

    if (
      Number.isInteger(versionNumber) &&
      versionNumber > maxVersion
    ) {
      maxVersion = versionNumber;
    }
  }


  const version = maxVersion + 1;


  const versionTag =
    String(version).padStart(3, '0');


  /* -------------------------------------------------------
   * Resolve binary metadata
   * -----------------------------------------------------*/

  const binaryInfo =
    resolveBinaryInfo(
      item,
      binaryField
    );


  const extension =
    safePart(
      input.extension ||
      getBinaryExtension(binaryInfo)
    ).toLowerCase();


  if (!extension) {
    throw new Error(
      `Save Artifact: cannot determine file extension for binary field "${binaryField}"`
    );
  }


  const mime_type =
    cleanString(
      input.mime_type ||
      getMimeType(binaryInfo)
    ) ||
    'application/octet-stream';


  const source_filename =
    cleanString(
      input.source_filename ||
      binaryInfo.fileName ||
      binaryInfo.filename
    ) ||
    null;


  /* -------------------------------------------------------
   * Build filename
   *
   * Examples:
   *
   * run_x.audio.00.preview.001.flac
   * run_x.audio.00.preview.002.flac
   *
   * run_x.audio.00.full.001.flac
   * run_x.audio.00.full.002.flac
   *
   * run_x.lyrics.001.txt
   * run_x.image.001.png
   * -----------------------------------------------------*/

  const variantPart =
    variant
      ? `.${safePart(variant)}`
      : '';


  const modePart =
    mode
      ? `.${safePart(mode)}`
      : '';


  const filename =
    `${run_id}.${artifact_name}` +
    `${variantPart}` +
    `${modePart}` +
    `.${versionTag}.${extension}`;


  const filePath =
    joinPath(
      runDir,
      filename
    );


  /* -------------------------------------------------------
   * Collision protection
   * -----------------------------------------------------*/

  if (fs.existsSync(filePath)) {
    throw new Error(
      `Save Artifact: refusing to overwrite existing file "${filePath}"`
    );
  }


  /* -------------------------------------------------------
   * Read binary
   *
   * IMPORTANT:
   *
   * Do NOT use:
   *
   *   binaryInfo.data
   *
   * n8n may store binary data externally.
   *
   * getBinaryDataBuffer() works with n8n's binary
   * storage implementation.
   * -----------------------------------------------------*/

  const buffer =
    await this.helpers.getBinaryDataBuffer(
      itemIndex,
      binaryField
    );


  if (
    !buffer ||
    buffer.length === 0
  ) {
    throw new Error(
      `Save Artifact: binary field "${binaryField}" has no data`
    );
  }


  /* -------------------------------------------------------
   * Write artifact
   * -----------------------------------------------------*/

  fs.writeFileSync(
    filePath,
    buffer
  );


  /* -------------------------------------------------------
   * Build artifact record
   * -----------------------------------------------------*/

  const reference =
    normalizeReference(
      input.reference
    );


  const created_at =
    new Date().toISOString();


  const artifactRecord = {
    name: artifact_name,
    variant,
    mode,
    version,
    filename,
    path: filePath,
    mime_type,
    size_bytes: buffer.length,
    created_at,
    reference
  };


  /*
   * Preserve useful metadata when supplied.
   */

  const optionalFields = [
    'template_filename',
    'music_template_filename',
    'seed',
    'lyrics_version',
    'source_filename'
  ];


  for (const field of optionalFields) {
    if (
      input[field] !== undefined &&
      input[field] !== null &&
      input[field] !== ''
    ) {
      artifactRecord[field] =
        input[field];
    }
  }


  /*
   * source_filename:
   * use resolved binary filename if it wasn't
   * explicitly supplied in input.
   */
  if (
    !artifactRecord.source_filename &&
    source_filename
  ) {
    artifactRecord.source_filename =
      source_filename;
  }


  /* -------------------------------------------------------
   * Update manifest
   * -----------------------------------------------------*/

  manifestData.artifacts.push(
    artifactRecord
  );


  /*
   * Keep manifest metadata minimal.
   */
  manifestData.run_id = run_id;
  manifestData.updated_at =
    created_at;


  writeJsonFile(
    manifestPath,
    manifestData
  );


  /* -------------------------------------------------------
   * Return
   * -----------------------------------------------------*/

  output.push({
    json: {
      ...input,

      run_id,

      production_root:
        productionRoot,

      run_dir:
        runDir,

      manifest_path:
        manifestPath,

      artifact_name,

      variant,

      mode,

      artifact_version:
        version,

      version,

      filename,

      path:
        filePath,

      mime_type,

      size_bytes:
        buffer.length,

      created_at,

      reference,

      source_filename,

      saved_artifact:
        artifactRecord
    },
    binary: {
      data: await this.helpers.prepareBinaryData(
        buffer,
        filename,
        mime_type
      )
    }
  });
}


return output;
