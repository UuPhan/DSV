// ============================================================
// Save Artifact
//
// Purpose:
// - Save binary artifacts to the production filesystem.
// - Register already-existing filesystem artifacts without
//   copying them again.
// - Maintain artifact manifest.
// - Version scope:
//
//     name + variant + mode
//
// Binary flow:
//
//     binary exists
//         ↓
//     create canonical artifact filename
//         ↓
//     write binary
//         ↓
//     register manifest
//
// Existing-file flow:
//
//     no binary
//         ↓
//     resolve input.path / input.output_path
//         ↓
//     file exists
//         ↓
//     register manifest only
//
// If neither binary nor existing file exists:
//     throw error
//
// Canonical filename examples:
//
//     run_xxx.lyrics.001.txt
//     run_xxx.image.001.png
//     run_xxx.audio.00.preview.001.flac
//     run_xxx.audio.00.full.001.flac
//     run_xxx.video.002.mp4
//     run_xxx.video.visual.001.mp4
// ============================================================

const fs = require('fs');


// ============================================================
// 1. Helpers
// ============================================================

function cleanString(value) {
  if (
    value === null ||
    value === undefined
  ) {
    return '';
  }

  return String(value).trim();
}


function safePart(value) {
  const result =
    cleanString(value)
      .replace(/[^a-zA-Z0-9._-]+/g, '_')
      .replace(/^_+|_+$/g, '');

  if (!result) {
    throw new Error(
      `Save Artifact: invalid path component: "${value}"`
    );
  }

  return result;
}


function joinPath(...parts) {
  if (!parts.length) {
    return '';
  }

  const first =
    String(
      parts[0] ?? ''
    );

  const isAbsolute =
    first.startsWith('/');

  const cleaned =
    parts
      .map(part =>
        String(
          part ?? ''
        )
          .replace(
            /^\/+|\/+$/g,
            ''
          )
      )
      .filter(Boolean)
      .join('/');

  return isAbsolute
    ? `/${cleaned}`
    : cleaned;
}


function normalizeReference(value) {
  if (
    value === null ||
    value === undefined
  ) {
    return [];
  }

  const values =
    Array.isArray(value)
      ? value
      : [value];

  return values
    .map(item =>
      cleanString(item)
    )
    .filter(Boolean);
}


function normalizeAbsolutePath(value) {
  const raw =
    cleanString(value);

  if (!raw) {
    return '';
  }

  if (raw.startsWith('/')) {
    return raw;
  }

  return `/data/${raw}`;
}


function mimeFromExtension(extension) {
  const map = {
    txt: 'text/plain',
    json: 'application/json',

    png: 'image/png',
    jpg: 'image/jpeg',
    jpeg: 'image/jpeg',
    webp: 'image/webp',

    mp3: 'audio/mpeg',
    wav: 'audio/wav',
    flac: 'audio/flac',
    ogg: 'audio/ogg',
    m4a: 'audio/mp4',

    mp4: 'video/mp4',
    webm: 'video/webm',
    mov: 'video/quicktime',
    mkv: 'video/x-matroska'
  };

  return (
    map[
      cleanString(
        extension
      ).toLowerCase()
    ] ||
    null
  );
}


function extensionFromFilename(filename) {
  const match =
    cleanString(
      filename
    ).match(
      /\.([a-zA-Z0-9]+)$/
    );

  return match
    ? match[1].toLowerCase()
    : '';
}


function extensionFromMime(mimeType) {
  const mime =
    cleanString(
      mimeType
    ).toLowerCase();

  const map = {
    'text/plain': 'txt',
    'application/json': 'json',

    'image/png': 'png',
    'image/jpeg': 'jpg',
    'image/webp': 'webp',

    'audio/mpeg': 'mp3',
    'audio/wav': 'wav',
    'audio/x-wav': 'wav',
    'audio/flac': 'flac',
    'audio/ogg': 'ogg',
    'audio/mp4': 'm4a',

    'video/mp4': 'mp4',
    'video/webm': 'webm',
    'video/quicktime': 'mov',
    'video/x-matroska': 'mkv'
  };

  return (
    map[mime] ||
    ''
  );
}


// ============================================================
// 2. Resolve binary information
//
// n8n 2.x may store binary data externally.
// Do NOT inspect binaryData.data directly.
//
// The presence of the binary property is enough to enter
// binary mode. Actual bytes are loaded with:
// this.helpers.getBinaryDataBuffer(...)
// ============================================================

function resolveBinaryData(item) {
  if (
    !item ||
    !item.binary
  ) {
    return null;
  }

  const binary =
    item.binary;

  const fields =
    Object.keys(
      binary
    );

  if (!fields.length) {
    return null;
  }

  const preferred =
    fields.includes('data')
      ? 'data'
      : fields[0];

  const binaryInfo =
    binary[preferred];

  if (!binaryInfo) {
    return null;
  }

  return {
    field: preferred,
    data: binaryInfo
  };
}


// ============================================================
// 3. Current input
// ============================================================

const items =
  $input.all();

if (!items.length) {
  throw new Error(
    'Save Artifact: no input items.'
  );
}


// ============================================================
// 4. Production root
// ============================================================

const DEFAULT_PRODUCTION_ROOT =
  '/data/productions';


// ============================================================
// 5. Process items
// ============================================================

const output = [];

for (
  let itemIndex = 0;
  itemIndex < items.length;
  itemIndex++
) {
  const item =
    items[itemIndex];

  const input =
    {
      ...(item.json || {})
    };


  // ==========================================================
  // 5.1 run_id
  // ==========================================================

  const run_id =
    cleanString(
      input.run_id
    );

  if (!run_id) {
    throw new Error(
      'Save Artifact: missing run_id.'
    );
  }

  if (
    run_id.includes('/') ||
    run_id.includes('\\') ||
    run_id.includes('..')
  ) {
    throw new Error(
      `Save Artifact: invalid run_id: ${run_id}`
    );
  }


  // ==========================================================
  // 5.2 Production root
  // ==========================================================

  const productionRoot =
    cleanString(
      input.production_root
    ) ||
    DEFAULT_PRODUCTION_ROOT;

  if (
    !productionRoot.startsWith('/')
  ) {
    throw new Error(
      `Save Artifact: production_root must be absolute: ${productionRoot}`
    );
  }


  // ==========================================================
  // 5.3 Run directory
  // ==========================================================

  const runDir =
    joinPath(
      productionRoot,
      run_id
    );


  // ==========================================================
  // 5.4 Manifest
  // ==========================================================

  const manifestPath =
    cleanString(
      input.manifest_path
    ) ||
    joinPath(
      runDir,
      `${run_id}.artifacts.json`
    );


  // ==========================================================
  // 5.5 Artifact identity
  // ==========================================================

  const artifact_name =
    cleanString(
      input.artifact_name ||
      input.name
    );

  if (!artifact_name) {
    throw new Error(
      `Save Artifact: missing artifact_name for run_id: ${run_id}`
    );
  }

  const variant =
    cleanString(
      input.variant
    ) || null;

  const mode =
    cleanString(
      input.mode
    ).toLowerCase() || null;


  // ==========================================================
  // 5.6 Extension / MIME from input
  // ==========================================================

  let extension =
    cleanString(
      input.extension
    ).toLowerCase();

  let mime_type =
    cleanString(
      input.mime_type
    );


  // ==========================================================
  // 5.7 Load existing manifest
  // ==========================================================

  let manifest = {
    run_id,
    artifacts: []
  };

  if (
    fs.existsSync(
      manifestPath
    )
  ) {
    const raw =
      fs.readFileSync(
        manifestPath,
        'utf8'
      );

    if (raw.trim()) {
      try {
        manifest =
          JSON.parse(
            raw
          );
      } catch (error) {
        throw new Error(
          `Save Artifact: invalid manifest JSON: ${manifestPath}`
        );
      }
    }
  }

  if (
    !manifest ||
    typeof manifest !== 'object'
  ) {
    manifest = {
      run_id,
      artifacts: []
    };
  }

  if (
    !Array.isArray(
      manifest.artifacts
    )
  ) {
    manifest.artifacts = [];
  }

  manifest.run_id =
    run_id;


  // ==========================================================
  // 5.8 Determine next version
  //
  // Version scope:
  //
  //     name + variant + mode
  //
  // Example:
  //
  // audio / 00 / preview
  // audio / 00 / full
  //
  // are independent version sequences.
  // ==========================================================

  const scopedArtifacts =
    manifest.artifacts
      .filter(artifact => {
        if (
          !artifact ||
          artifact.name !== artifact_name
        ) {
          return false;
        }

        const artifactVariant =
          cleanString(
            artifact.variant
          ) || null;

        const artifactMode =
          cleanString(
            artifact.mode
          ).toLowerCase() || null;

        return (
          artifactVariant === variant &&
          artifactMode === mode
        );
      });

  const maxVersion =
    scopedArtifacts.reduce(
      (max, artifact) => {
        const value =
          Number(
            artifact.version
          );

        return (
          Number.isInteger(value) &&
          value > max
        )
          ? value
          : max;
      },
      0
    );

  const version =
    maxVersion + 1;

  const versionTag =
    String(
      version
    ).padStart(
      3,
      '0'
    );


  // ==========================================================
  // 5.9 Normalize reference
  // ==========================================================

  const reference =
    normalizeReference(
      input.reference ??
      input.references ??
      input.saved_artifact?.reference ??
      input.artifact?.reference ??
      []
    );


  // ==========================================================
  // 5.10 Detect binary
  // ==========================================================

  const binaryInfo =
    resolveBinaryData(
      item
    );


  // ==========================================================
  // 5.11 Variables
  // ==========================================================

  let filename = null;
  let filePath = null;
  let size_bytes = null;


  // ==========================================================
  // 6. BINARY MODE
  //
  // Binary exists:
  //
  // - generate canonical filename
  // - save binary
  // - register artifact
  // ==========================================================

  if (binaryInfo) {

    const binaryData =
      binaryInfo.data;


    // --------------------------------------------------------
    // Resolve MIME
    // --------------------------------------------------------

    mime_type =
      mime_type ||
      cleanString(
        binaryData.mimeType
      ) ||
      null;


    // --------------------------------------------------------
    // Resolve extension
    // --------------------------------------------------------

    const binaryFilename =
      cleanString(
        binaryData.fileName
      );

    if (!extension) {
      extension =
        extensionFromFilename(
          binaryFilename
        );
    }

    if (!extension) {
      extension =
        extensionFromMime(
          mime_type
        );
    }

    if (!extension) {
      extension =
        'bin';
    }


    // --------------------------------------------------------
    // Canonical filename
    //
    // name.variant.mode.version.ext
    //
    // Empty variant/mode fields are omitted.
    // --------------------------------------------------------

    const variantPart =
      variant
        ? `.${safePart(variant)}`
        : '';

    const modePart =
      mode
        ? `.${safePart(mode)}`
        : '';

    filename =
      `${safePart(run_id)}` +
      `.${safePart(artifact_name)}` +
      `${variantPart}` +
      `${modePart}` +
      `.${versionTag}` +
      `.${safePart(extension)}`;


    filePath =
      joinPath(
        runDir,
        filename
      );


    // --------------------------------------------------------
    // Prevent overwrite
    // --------------------------------------------------------

    if (
      fs.existsSync(
        filePath
      )
    ) {
      throw new Error(
        `Save Artifact: refusing to overwrite existing file "${filePath}"`
      );
    }


    // --------------------------------------------------------
    // Get binary bytes
    //
    // Required because n8n 2.x can store binary externally.
    // --------------------------------------------------------

    const buffer =
      await this.helpers.getBinaryDataBuffer(
        itemIndex,
        binaryInfo.field
      );

    if (
      !buffer ||
      buffer.length === 0
    ) {
      throw new Error(
        `Save Artifact: binary field "${binaryInfo.field}" has no data`
      );
    }


    // --------------------------------------------------------
    // Ensure directory
    // --------------------------------------------------------

    fs.mkdirSync(
      runDir,
      {
        recursive: true
      }
    );


    // --------------------------------------------------------
    // Write binary
    // --------------------------------------------------------

    fs.writeFileSync(
      filePath,
      buffer
    );

    size_bytes =
      buffer.length;


    // --------------------------------------------------------
    // Resolve MIME fallback
    // --------------------------------------------------------

    if (!mime_type) {
      mime_type =
        mimeFromExtension(
          extension
        );
    }
  }


  // ==========================================================
  // 7. EXISTING FILE MODE
  //
  // No binary.
  //
  // Producer already created the file:
  //
  //     FFmpeg
  //     ComfyUI
  //     external processor
  //
  // Save Artifact only registers it.
  // ==========================================================

  else {

    const existingPathRaw =
      cleanString(
        input.path ||
        input.output_path ||
        input.output?.path ||
        input.output?.output_path
      );

    if (!existingPathRaw) {
      throw new Error(
        `Save Artifact: no binary data and no existing artifact path for run_id: ${run_id}`
      );
    }


    // --------------------------------------------------------
    // Normalize path
    // --------------------------------------------------------

    filePath =
      normalizeAbsolutePath(
        existingPathRaw
      );


    // --------------------------------------------------------
    // Security check
    //
    // Existing artifacts must remain inside production root.
    // --------------------------------------------------------

    const normalizedRoot =
      productionRoot.endsWith('/')
        ? productionRoot
        : `${productionRoot}/`;

    if (
      !filePath.startsWith(
        normalizedRoot
      )
    ) {
      throw new Error(
        `Save Artifact: existing artifact path is outside production root: ${filePath}`
      );
    }


    // --------------------------------------------------------
    // File must already exist
    // --------------------------------------------------------

    if (
      !fs.existsSync(
        filePath
      )
    ) {
      throw new Error(
        `Save Artifact: existing artifact file does not exist: ${filePath}`
      );
    }


    // --------------------------------------------------------
    // Must be a regular file
    // --------------------------------------------------------

    const stat =
      fs.statSync(
        filePath
      );

    if (
      !stat.isFile()
    ) {
      throw new Error(
        `Save Artifact: existing artifact path is not a file: ${filePath}`
      );
    }


    // --------------------------------------------------------
    // Register existing file
    // --------------------------------------------------------

    filename =
      filePath
        .split('/')
        .pop();

    if (!filename) {
      throw new Error(
        `Save Artifact: cannot resolve filename from path: ${filePath}`
      );
    }

    size_bytes =
      stat.size;


    // --------------------------------------------------------
    // Resolve extension
    // --------------------------------------------------------

    if (!extension) {
      extension =
        extensionFromFilename(
          filename
        );
    }


    // --------------------------------------------------------
    // Resolve MIME
    // --------------------------------------------------------

    if (!mime_type) {
      mime_type =
        mimeFromExtension(
          extension
        );
    }


    // --------------------------------------------------------
    // IMPORTANT
    //
    // Existing file is NOT copied.
    // Existing file is NOT renamed.
    // Existing file is NOT overwritten.
    //
    // filePath remains the producer's actual path.
    // --------------------------------------------------------
  }


  // ==========================================================
  // 8. created_at
  // ==========================================================

  const created_at =
    new Date()
      .toISOString();


  // ==========================================================
  // 9. Artifact record
  // ==========================================================

  const artifactRecord = {
    name:
      artifact_name,

    variant:
      variant,

    mode:
      mode,

    version:
      version,

    filename:
      filename,

    path:
      filePath,

    mime_type:
      mime_type ||
      null,

    size_bytes:
      size_bytes,

    created_at:
      created_at,

    reference:
      reference
  };


  // ==========================================================
  // 10. Preserve additional metadata
  //
  // These are useful for audio/video/image provenance.
  // ==========================================================

  const metadataFields = [
    'seed',
    'lyrics_version',
    'music_template_filename',
    'source_filename',
    'output_filename',
    'duration_seconds',
    'width',
    'height',
    'fps',
    'title',
    'channel_name',
    'visual_direction',
    'theme',
    'core_message'
  ];

  for (
    const field of metadataFields
  ) {
    if (
      input[field] !== undefined &&
      input[field] !== null
    ) {
      artifactRecord[field] =
        input[field];
    }
  }


  // ==========================================================
  // 11. Preserve arbitrary artifact metadata
  //
  // If upstream explicitly provides saved_artifact metadata,
  // merge only fields that do not conflict with canonical
  // artifact identity.
  // ==========================================================

  if (
    input.saved_artifact &&
    typeof input.saved_artifact === 'object'
  ) {
    const ignoredFields = new Set([
      'name',
      'artifact_name',
      'variant',
      'mode',
      'version',
      'filename',
      'path',
      'mime_type',
      'size_bytes',
      'created_at',
      'reference'
    ]);

    for (
      const [key, value]
      of Object.entries(
        input.saved_artifact
      )
    ) {
      if (
        ignoredFields.has(key)
      ) {
        continue;
      }

      if (
        artifactRecord[key] === undefined
      ) {
        artifactRecord[key] =
          value;
      }
    }
  }


  // ==========================================================
  // 12. Append manifest
  // ==========================================================

  manifest.artifacts.push(
    artifactRecord
  );


  // ==========================================================
  // 13. Ensure manifest directory
  // ==========================================================

  fs.mkdirSync(
    runDir,
    {
      recursive: true
    }
  );


  // ==========================================================
  // 14. Write manifest
  // ==========================================================

  fs.writeFileSync(
    manifestPath,
    JSON.stringify(
      manifest,
      null,
      2
    ),
    'utf8'
  );


  // ==========================================================
  // 15. Return result
  //
  // IMPORTANT:
  //
  // Existing-file mode intentionally returns NO binary.
  //
  // Binary mode returns binary so downstream Telegram nodes
  // can still send the newly saved artifact when needed.
  // ==========================================================

  const resultJson = {
    ...input,

    run_id:
      run_id,

    production_root:
      productionRoot,

    run_dir:
      runDir,

    manifest_path:
      manifestPath,

    artifact_name:
      artifact_name,

    variant:
      variant,

    mode:
      mode,

    artifact_version:
      version,

    version:
      version,

    filename:
      filename,

    path:
      filePath,

    mime_type:
      mime_type ||
      null,

    size_bytes:
      size_bytes,

    created_at:
      created_at,

    reference:
      reference,

    saved_artifact:
      artifactRecord
  };


  // ==========================================================
  // 16. Return binary only when original input had binary
  // ==========================================================

  if (binaryInfo) {
    const binaryData =
      await this.helpers.getBinaryDataBuffer(
        itemIndex,
        binaryInfo.field
      );

    const preparedBinary =
      await this.helpers.prepareBinaryData(
        binaryData,
        filename,
        mime_type
      );

    output.push({
      json:
        resultJson,

      binary: {
        data:
          preparedBinary
      }
    });
  } else {
    output.push({
      json:
        resultJson
    });
  }
}


// ============================================================
// 17. Return
// ============================================================

return output;
